-- =========================================================
-- 6. CHECK PRODUCT INVENTORY
-- =========================================================

DROP PROCEDURE IF EXISTS check_product_inventory;

DELIMITER $$

CREATE PROCEDURE check_product_inventory(
    IN p_product_id INT
)
BEGIN

    SELECT
        p.product_id,
        p.product_name,
        w.warehouse_id,
        w.warehouse_name,
        i.quantity_available,
        i.quantity_reserved,
        (
            i.quantity_available -
            i.quantity_reserved
        ) AS free_quantity,
        p.reorder_level,

        CASE
            WHEN i.quantity_available = 0
                THEN 'OUT OF STOCK'

            WHEN i.quantity_available < p.reorder_level
                THEN 'REORDER REQUIRED'

            ELSE 'SUFFICIENT STOCK'
        END AS stock_status

    FROM inventory i

    JOIN products p
        ON i.product_id = p.product_id

    JOIN warehouses w
        ON i.warehouse_id = w.warehouse_id

    WHERE i.product_id = p_product_id;

END $$

DELIMITER ;


CALL check_product_inventory(1);



-- =========================================================
-- 7. GET TOTAL PRODUCT STOCK
-- =========================================================

DROP PROCEDURE IF EXISTS get_total_product_stock;

DELIMITER $$

CREATE PROCEDURE get_total_product_stock(
    IN p_product_id INT
)
BEGIN

    SELECT
        p.product_id,
        p.product_name,
        COALESCE(
            SUM(i.quantity_available),
            0
        ) AS total_stock,
        COALESCE(
            SUM(i.quantity_reserved),
            0
        ) AS total_reserved,
        COALESCE(
            SUM(
                i.quantity_available -
                i.quantity_reserved
            ),
            0
        ) AS free_stock
    FROM products p
    LEFT JOIN inventory i
        ON p.product_id = i.product_id
    WHERE p.product_id = p_product_id
    GROUP BY
        p.product_id,
        p.product_name;

END $$

DELIMITER ;


CALL get_total_product_stock(1);



-- =========================================================
-- 8. GET LOW STOCK PRODUCTS
-- =========================================================

DROP PROCEDURE IF EXISTS get_low_stock_products;

DELIMITER $$

CREATE PROCEDURE get_low_stock_products()
BEGIN

    SELECT
        p.product_id,
        p.product_name,
        p.reorder_level,
        COALESCE(
            SUM(i.quantity_available),
            0
        ) AS total_stock
    FROM products p
    LEFT JOIN inventory i
        ON p.product_id = i.product_id
    GROUP BY
        p.product_id,
        p.product_name,
        p.reorder_level
    HAVING total_stock < p.reorder_level
    ORDER BY total_stock;

END $$

-- 10. GET PRODUCT SALES
-- =========================================================

DROP PROCEDURE IF EXISTS get_product_sales;

DELIMITER $$

CREATE PROCEDURE get_product_sales(
    IN p_product_id INT
)
BEGIN

    SELECT
        p.product_id,
        p.product_name,
        SUM(oi.quantity) AS units_sold,
        ROUND(
            SUM(oi.subtotal),
            2
        ) AS total_sales
    FROM products p
    JOIN order_items oi
        ON p.product_id = oi.product_id
    JOIN orders o
        ON oi.order_id = o.order_id
    WHERE p.product_id = p_product_id
      AND o.order_status NOT IN (
          'CANCELLED',
          'RETURNED'
      )
    GROUP BY
        p.product_id,
        p.product_name;

END $$

DELIMITER ;


CALL get_product_sales(1);



-- =========================================================
-- 11. GET SALES BETWEEN DATES
-- =========================================================

DROP PROCEDURE IF EXISTS get_sales_between_dates;

DELIMITER $$

CREATE PROCEDURE get_sales_between_dates(
    IN p_start_date DATE,
    IN p_end_date DATE
)
BEGIN

    SELECT
        COUNT(*) AS total_orders,
        ROUND(
            SUM(total_amount),
            2
        ) AS total_sales,
        ROUND(
            AVG(total_amount),
            2
        ) AS average_order_value
    FROM orders
    WHERE DATE(order_date)
          BETWEEN p_start_date AND p_end_date
      AND order_status NOT IN (
          'CANCELLED',
          'RETURNED'
      );

END $$

DELIMITER ;

    '2026-01-01',
    '2026-12-31'
);

-- 12. GET CUSTOMER SPENDING
-- =========================================================

DROP PROCEDURE IF EXISTS get_customer_spending;

DELIMITER $$

CREATE PROCEDURE get_customer_spending(
    IN p_customer_id INT
)
BEGIN

    SELECT
        c.customer_id,
        CONCAT(
            c.first_name,
            ' ',
            c.last_name
        ) AS customer_name,

        COUNT(o.order_id)
            AS total_orders,

        COALESCE(
            ROUND(
                SUM(o.total_amount),
                2
            ),
            0
        ) AS total_spending,

        COALESCE(
            ROUND(
                AVG(o.total_amount),
                2
            ),
            0
        ) AS average_order_value

    FROM customers c

    LEFT JOIN orders o
        ON c.customer_id = o.customer_id
       AND o.order_status NOT IN (
           'CANCELLED',
           'RETURNED'
       )

    WHERE c.customer_id = p_customer_id

    GROUP BY
        c.customer_id,
        c.first_name,
        c.last_name;

END $$

DELIMITER ;


CALL get_customer_spending(1);

-- 25. RESERVE INVENTORY
-- =========================================================

DROP PROCEDURE IF EXISTS reserve_inventory;

DELIMITER $$

CREATE PROCEDURE reserve_inventory(
    IN p_product_id INT,
    IN p_warehouse_id INT,
    IN p_quantity INT
)
BEGIN

    DECLARE v_available INT DEFAULT 0;

    IF p_quantity <= 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Quantity must be greater than zero';

    END IF;

    SELECT
        quantity_available -
        quantity_reserved

    INTO v_available

    FROM inventory

    WHERE product_id = p_product_id
      AND warehouse_id = p_warehouse_id

    FOR UPDATE;

    IF v_available IS NULL THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Product inventory not found';

    ELSEIF v_available < p_quantity THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Insufficient available inventory';

    ELSE

        UPDATE inventory

        SET quantity_reserved =
            quantity_reserved + p_quantity

        WHERE product_id = p_product_id
          AND warehouse_id = p_warehouse_id;

        SELECT
            'Inventory reserved successfully'
                AS message,

            quantity_available,
            quantity_reserved,
            (
                quantity_available -
                quantity_reserved
            ) AS free_quantity

        FROM inventory

        WHERE product_id = p_product_id
          AND warehouse_id = p_warehouse_id;

    END IF;

END $$

DELIMITER ;

-- =========================================================
-- 26. RELEASE RESERVED INVENTORY
-- =========================================================

DROP PROCEDURE IF EXISTS release_reserved_inventory;

DELIMITER $$

CREATE PROCEDURE release_reserved_inventory(
    IN p_product_id INT,
    IN p_warehouse_id INT,
    IN p_quantity INT
)
BEGIN

    DECLARE v_reserved INT DEFAULT 0;

    IF p_quantity <= 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Quantity must be greater than zero';

    END IF;

    SELECT quantity_reserved
    INTO v_reserved

    FROM inventory

    WHERE product_id = p_product_id
      AND warehouse_id = p_warehouse_id

    FOR UPDATE;

    IF v_reserved IS NULL THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Inventory record not found';

    ELSEIF v_reserved < p_quantity THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Release quantity exceeds reserved quantity';

    ELSE

        UPDATE inventory

        SET quantity_reserved =
            quantity_reserved - p_quantity

        WHERE product_id = p_product_id
          AND warehouse_id = p_warehouse_id;

        SELECT
            'Reserved inventory released successfully'
                AS message;

    END IF;

END $$

DELIMITER ;

-- =========================================================
-- 27. RESTOCK INVENTORY
-- =========================================================

DROP PROCEDURE IF EXISTS restock_inventory;

DELIMITER $$

CREATE PROCEDURE restock_inventory(
    IN p_product_id INT,
    IN p_warehouse_id INT,
    IN p_quantity INT
)
BEGIN

    IF p_quantity <= 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Restock quantity must be greater than zero';

    END IF;

    UPDATE inventory

    SET
        quantity_available =
            quantity_available + p_quantity,

        last_restocked_at = NOW()

    WHERE product_id = p_product_id
      AND warehouse_id = p_warehouse_id;

    IF ROW_COUNT() = 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Inventory record not found';

    END IF;

    INSERT INTO inventory_transactions
    (
        product_id,
        warehouse_id,
        transaction_type,
        quantity,
        reference_type,
        reference_id,
        transaction_date,
        remarks
    )
    VALUES
    (
        p_product_id,
        p_warehouse_id,
        'RESTOCK',
        p_quantity,
        'MANUAL',
        NULL,
        NOW(),
        'Inventory restocked using stored procedure'
    );

    SELECT
        'Inventory restocked successfully'
            AS message;

END $$

DELIMITER ;

-- 28. CREATE ORDER
-- =========================================================

DROP PROCEDURE IF EXISTS create_order;

DELIMITER $$

CREATE PROCEDURE create_order(
    IN p_customer_id INT,
    IN p_shipping_address_id INT,
    IN p_subtotal DECIMAL(12,2),
    IN p_discount_amount DECIMAL(12,2),
    IN p_tax_amount DECIMAL(12,2),
    IN p_shipping_charge DECIMAL(12,2)
)
BEGIN

    DECLARE v_total DECIMAL(12,2);

    IF p_subtotal < 0
       OR p_discount_amount < 0
       OR p_tax_amount < 0
       OR p_shipping_charge < 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Order amounts cannot be negative';

    END IF;

    SET v_total =
        p_subtotal
        - p_discount_amount
        + p_tax_amount
        + p_shipping_charge;

    INSERT INTO orders
    (
        customer_id,
        order_date,
        order_status,
        shipping_address_id,
        subtotal,
        discount_amount,
        tax_amount,
        shipping_charge,
        total_amount
    )
    VALUES
    (
        p_customer_id,
        NOW(),
        'PROCESSING',
        p_shipping_address_id,
        p_subtotal,
        p_discount_amount,
        p_tax_amount,
        p_shipping_charge,
        v_total
    );

    SELECT
        LAST_INSERT_ID() AS order_id,
        v_total AS total_amount,
        'Order created successfully'
            AS message;

END $$

DELIMITER ;

-- =========================================================
-- 29. ADD ORDER ITEM
-- =========================================================

DROP PROCEDURE IF EXISTS add_order_item;

DELIMITER $$

CREATE PROCEDURE add_order_item(
    IN p_order_id BIGINT,
    IN p_product_id INT,
    IN p_quantity INT,
    IN p_unit_price DECIMAL(12,2),
    IN p_discount_amount DECIMAL(12,2)
)
BEGIN

    DECLARE v_subtotal DECIMAL(12,2);

    IF p_quantity <= 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Quantity must be greater than zero';

    END IF;

    IF p_unit_price < 0
       OR p_discount_amount < 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Price or discount cannot be negative';

    END IF;

    SET v_subtotal =
        (p_quantity * p_unit_price)
        - p_discount_amount;

    IF v_subtotal < 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Discount cannot exceed item value';

    END IF;

    INSERT INTO order_items
    (
        order_id,
        product_id,
        quantity,
        unit_price,
        discount_amount,
        subtotal
    )
    VALUES
    (
        p_order_id,
        p_product_id,
        p_quantity,
        p_unit_price,
        p_discount_amount,
        v_subtotal
    );

    SELECT
        LAST_INSERT_ID() AS order_item_id,
        v_subtotal AS item_subtotal,
        'Order item added successfully'
            AS message;

END $$

DELIMITER ;

-- 30. UPDATE ORDER TOTAL
-- =========================================================

DROP PROCEDURE IF EXISTS update_order_total;

DELIMITER $$

CREATE PROCEDURE update_order_total(
    IN p_order_id BIGINT
)
BEGIN

    DECLARE v_subtotal DECIMAL(12,2);
    DECLARE v_discount DECIMAL(12,2);
    DECLARE v_tax DECIMAL(12,2);
    DECLARE v_shipping DECIMAL(12,2);

    SELECT
        COALESCE(SUM(subtotal), 0)
    INTO v_subtotal
    FROM order_items
    WHERE order_id = p_order_id;

    SELECT
        discount_amount,
        tax_amount,
        shipping_charge
    INTO
        v_discount,
        v_tax,
        v_shipping
    FROM orders
    WHERE order_id = p_order_id;

    UPDATE orders
    SET
        subtotal = v_subtotal,
        total_amount =
            v_subtotal
            - COALESCE(v_discount, 0)
            + COALESCE(v_tax, 0)
            + COALESCE(v_shipping, 0)
    WHERE order_id = p_order_id;

    SELECT
        order_id,
        subtotal,
        discount_amount,
        tax_amount,
        shipping_charge,
        total_amount
    FROM orders
    WHERE order_id = p_order_id;

END $$

DELIMITER ;

-- =========================================================
-- 31. PROCESS PAYMENT
-- =========================================================

DROP PROCEDURE IF EXISTS process_payment;

DELIMITER $$

CREATE PROCEDURE process_payment(
    IN p_order_id BIGINT,
    IN p_payment_method VARCHAR(30),
    IN p_transaction_reference VARCHAR(100),
    IN p_amount DECIMAL(12,2)
)
BEGIN

    DECLARE v_order_total DECIMAL(12,2);
    DECLARE v_order_status VARCHAR(30);

    SELECT
        total_amount,
        order_status
    INTO
        v_order_total,
        v_order_status
    FROM orders
    WHERE order_id = p_order_id;

    IF v_order_total IS NULL THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Order not found';

    END IF;

    IF v_order_status = 'CANCELLED' THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Cannot process payment for cancelled order';

    END IF;

    IF p_amount <= 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Payment amount must be greater than zero';

    END IF;

    INSERT INTO payments
    (
        order_id,
        payment_date,
        payment_method,
        transaction_reference,
        amount,
        payment_status
    )
    VALUES
    (
        p_order_id,
        NOW(),
        p_payment_method,
        p_transaction_reference,
        p_amount,
        CASE
            WHEN p_amount >= v_order_total
                THEN 'SUCCESS'
            ELSE 'PENDING'
        END
    );

    SELECT
        LAST_INSERT_ID() AS payment_id,
        'Payment processed'
            AS message;

END $$

DELIMITER ;

    IN p_order_id BIGINT,
    IN p_return_reason VARCHAR(255),
    IN p_refund_amount DECIMAL(12,2)
)

    DECLARE v_order_exists INT DEFAULT 0;

    SELECT COUNT(*)
    INTO v_order_exists
    FROM orders
    WHERE order_id = p_order_id;

    IF v_order_exists = 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Order not found';

    END IF;

    IF p_refund_amount < 0 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Refund amount cannot be negative';

    END IF;

    INSERT INTO returns
    (
        order_id,
        return_date,
        return_reason,
        return_status,
        refund_amount
    )
    VALUES
    (
        p_order_id,
        NOW(),
        p_return_reason,
        'REQUESTED',
        p_refund_amount
    );

    SELECT
        LAST_INSERT_ID() AS return_id,
        'Return request created'
            AS message;

END $$

-- 41. GET ORDER SUMMARY
-- =========================================================

DROP PROCEDURE IF EXISTS get_order_summary;

DELIMITER $$

CREATE PROCEDURE get_order_summary(
    IN p_order_id BIGINT
)
BEGIN

    SELECT
        o.order_id,
        o.order_date,
        o.order_status,

        CONCAT(
            c.first_name,
            ' ',
            c.last_name
        ) AS customer_name,

        c.email,

        COUNT(oi.order_item_id)
            AS item_lines,

        COALESCE(
            SUM(oi.quantity),
            0
        ) AS total_items,

        o.subtotal,
        o.discount_amount,
        o.tax_amount,
        o.shipping_charge,
        o.total_amount,

        COALESCE(
            SUM(
                CASE
                    WHEN pay.payment_status = 'SUCCESS'
                    THEN pay.amount
                    ELSE 0
                END
            ),
            0
        ) AS successful_payment_amount

    FROM orders o

    JOIN customers c
        ON o.customer_id = c.customer_id

    LEFT JOIN order_items oi
        ON o.order_id = oi.order_id

    LEFT JOIN payments pay
        ON o.order_id = pay.order_id

    WHERE o.order_id = p_order_id

    GROUP BY
        o.order_id,
        o.order_date,
        o.order_status,
        c.first_name,
        c.last_name,
        c.email,
        o.subtotal,
        o.discount_amount,
        o.tax_amount,
        o.shipping_charge,
        o.total_amount;

END $$

DELIMITER ;

DELIMITER $$

CREATE PROCEDURE get_customer_lifetime_value(
    IN p_customer_id INT
)
BEGIN

    SELECT
        c.customer_id,

        CONCAT(
            c.first_name,
            ' ',

        ) AS customer_name,

        COUNT(o.order_id)
            AS total_orders,

        COALESCE(
            ROUND(

            ),
            0
        ) AS lifetime_value,

    LEFT JOIN orders o
        ON c.customer_id = o.customer_id
       AND o.order_status NOT IN (
           'CANCELLED',
           'RETURNED'
       )

    WHERE c.customer_id = p_customer_id

    GROUP BY

        c.first_name,
