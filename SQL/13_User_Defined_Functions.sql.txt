-- =========================================================
-- 1. CALCULATE PROFIT
-- =========================================================

DROP FUNCTION IF EXISTS calculate_profit;

DELIMITER $$

CREATE FUNCTION calculate_profit(
    p_cost_price DECIMAL(12,2),
    p_selling_price DECIMAL(12,2)
)
RETURNS DECIMAL(12,2)
DETERMINISTIC
BEGIN
    RETURN ROUND(
        p_selling_price - p_cost_price,
        2
    );
END $$

DELIMITER ;


-- Test
SELECT calculate_profit(1000, 1250) AS profit;


-- =========================================================
-- 2. CALCULATE PROFIT MARGIN
-- =========================================================

DROP FUNCTION IF EXISTS calculate_profit_margin;

DELIMITER $$

CREATE FUNCTION calculate_profit_margin(
    p_cost_price DECIMAL(12,2),
    p_selling_price DECIMAL(12,2)
)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN ROUND(
        (
            (p_selling_price - p_cost_price)
            / NULLIF(p_cost_price, 0)
        ) * 100,
        2
    );
END $$

DELIMITER ;


-- Test
SELECT calculate_profit_margin(1000, 1250)
AS profit_margin_percentage;

DELIMITER $$

CREATE FUNCTION calculate_delivery_days(
    p_shipped_date DATETIME,
    p_delivered_date DATETIME
)
RETURNS INT
DETERMINISTIC
BEGIN
    IF p_shipped_date IS NULL
       OR p_delivered_date IS NULL THEN
        RETURN NULL;
    END IF;

    RETURN DATEDIFF(
        p_delivered_date,
        p_shipped_date
    );
END $$

DROP FUNCTION IF EXISTS get_stock_status;

DELIMITER $$

CREATE FUNCTION get_stock_status(
    p_available_quantity INT,
    p_reorder_level INT
)
RETURNS VARCHAR(30)
DETERMINISTIC
BEGIN

    IF p_available_quantity IS NULL THEN
        RETURN 'UNKNOWN';

    ELSEIF p_available_quantity = 0 THEN
        RETURN 'OUT OF STOCK';

    ELSEIF p_available_quantity < p_reorder_level THEN
        RETURN 'REORDER REQUIRED';

    ELSE
        RETURN 'SUFFICIENT STOCK';
    END IF;

END $$

DELIMITER ;

-- 21. EMAIL DOMAIN
-- =========================================================

DROP FUNCTION IF EXISTS get_email_domain;

DELIMITER $$

CREATE FUNCTION get_email_domain(
    p_email VARCHAR(255)
)
RETURNS VARCHAR(100)
DETERMINISTIC
BEGIN

    IF p_email IS NULL
       OR LOCATE('@', p_email) = 0 THEN
        RETURN NULL;
    END IF;

    RETURN SUBSTRING_INDEX(
        LOWER(TRIM(p_email)),
        '@',
        -1
    );

END $$

DELIMITER ;

    'customer@gmail.com'
) AS email_domain;

CREATE FUNCTION is_valid_email(
    p_email VARCHAR(255)
)
RETURNS VARCHAR(20)
DETERMINISTIC
BEGIN

    IF p_email IS NULL
       OR TRIM(p_email) = '' THEN
        RETURN 'INVALID';

    ELSEIF p_email LIKE '%@%.%' THEN
        RETURN 'VALID';

    ELSE
        RETURN 'INVALID';
    END IF;

END $$

-- =========================================================
-- 23. COUPON DISCOUNT CALCULATION
-- =========================================================

DROP FUNCTION IF EXISTS calculate_coupon_discount;

DELIMITER $$

CREATE FUNCTION calculate_coupon_discount(
    p_order_amount DECIMAL(12,2),
    p_discount_type VARCHAR(20),
    p_discount_value DECIMAL(12,2),
    p_maximum_discount DECIMAL(12,2)
)
RETURNS DECIMAL(12,2)
DETERMINISTIC
BEGIN

    DECLARE v_discount DECIMAL(12,2);

    IF p_order_amount IS NULL
       OR p_order_amount <= 0 THEN
        RETURN 0;
    END IF;

    IF UPPER(p_discount_type) = 'PERCENTAGE' THEN

        SET v_discount =
            p_order_amount *
            p_discount_value / 100;

    ELSEIF UPPER(p_discount_type) = 'FIXED' THEN

        SET v_discount =
            p_discount_value;

    ELSE

        SET v_discount = 0;

    END IF;

    IF p_maximum_discount IS NOT NULL
       AND v_discount > p_maximum_discount THEN

        SET v_discount = p_maximum_discount;

    END IF;

    IF v_discount > p_order_amount THEN
        SET v_discount = p_order_amount;
    END IF;

    RETURN ROUND(v_discount, 2);

END $$

DELIMITER ;


-- Test
SELECT calculate_coupon_discount(
    10000,
    'PERCENTAGE',
    15,
    1000
) AS coupon_discount;


-- =========================================================
-- 24. FINAL ORDER AMOUNT AFTER COUPON
-- =========================================================

DROP FUNCTION IF EXISTS calculate_final_order_amount;

DELIMITER $$

CREATE FUNCTION calculate_final_order_amount(
    p_subtotal DECIMAL(12,2),
    p_coupon_discount DECIMAL(12,2),
    p_tax DECIMAL(12,2),
    p_shipping_charge DECIMAL(12,2)
)
RETURNS DECIMAL(12,2)
DETERMINISTIC
BEGIN

    RETURN ROUND(
        COALESCE(p_subtotal, 0)
        - COALESCE(p_coupon_discount, 0)
        + COALESCE(p_tax, 0)
        + COALESCE(p_shipping_charge, 0),
        2
    );

END $$

DELIMITER ;


-- Test

    20000,
    1500,
    3330,
    100
) AS final_order_amount;

-- =========================================================
-- 26. USE FUNCTIONS WITH INVENTORY
-- =========================================================

SELECT
    p.product_id,
    p.product_name,

    SUM(i.quantity_available)
        AS available_quantity,

    p.reorder_level,

    get_stock_status(
        SUM(i.quantity_available),
        p.reorder_level
    ) AS stock_status,

    calculate_inventory_value(
        SUM(i.quantity_available),
        p.cost_price
    ) AS inventory_value

FROM products p

JOIN inventory i
    ON p.product_id = i.product_id

GROUP BY
    p.product_id,
    p.product_name,
    p.reorder_level,
    p.cost_price

ORDER BY available_quantity;

-- 35. FUNCTION-BASED BUSINESS DASHBOARD
-- =========================================================

SELECT
    COUNT(*) AS total_products,

    ROUND(
        AVG(
            calculate_profit_margin(
                cost_price,
                selling_price
            )
        ),
        2
    ) AS average_profit_margin,

    ROUND(
        SUM(
            calculate_profit(
                cost_price,
                selling_price
            )
        ),
        2
    ) AS total_unit_profit,

    SUM(
        CASE
            WHEN get_price_category(selling_price)
                = 'PREMIUM'
            THEN 1
            ELSE 0
        END
    ) AS premium_products,

    SUM(
        CASE
            WHEN get_price_category(selling_price)
                = 'LOW'
            THEN 1
            ELSE 0
        END
    ) AS low_price_products

FROM products;
