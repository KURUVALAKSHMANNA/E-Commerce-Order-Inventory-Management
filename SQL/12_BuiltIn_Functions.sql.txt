-- 55. Inventory status

    i.inventory_id,
    i.product_id,
    i.quantity_available,
    p.reorder_level,

    ON i.product_id = p.product_id;

-- 80. Calculate expected final amount

    order_id,
    total_amount AS stored_total,
    ROUND(

    ) AS calculated_total,
    ROUND(
        total_amount -
        (
            subtotal
            - discount_amount
            + tax_amount
            + shipping_charge

        2
    ) AS difference
FROM orders;

-- =========================================================
-- 13. COALESCE + CASE BUSINESS REPORT
-- =========================================================

-- 88. Customer order summary
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COUNT(o.order_id) AS total_orders,
    COALESCE(
        ROUND(SUM(o.total_amount), 2),
        0
    ) AS total_spending,
    COALESCE(
        ROUND(AVG(o.total_amount), 2),
        0
    ) AS average_order_value,
    CASE
        WHEN COALESCE(SUM(o.total_amount), 0) >= 50000
            THEN 'PREMIUM'
        WHEN COALESCE(SUM(o.total_amount), 0) >= 25000
            THEN 'REGULAR'
        WHEN COUNT(o.order_id) > 0
            THEN 'OCCASIONAL'
        ELSE 'NO ORDERS'
    END AS customer_segment
FROM customers c
LEFT JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY total_spending DESC;

-- 14. DATE-BASED BUSINESS REPORTS

-- 89. Daily order summary

    DATE(order_date) AS order_day,
    COUNT(*) AS order_count,
    ROUND(SUM(total_amount), 2) AS daily_sales

-- 90. Weekly order summary
SELECT

    WEEK(order_date) AS order_week,

    YEAR(order_date),
    WEEK(order_date)

    order_week;


-- 91. Quarterly sales
SELECT

    QUARTER(order_date) AS order_quarter,

    ROUND(SUM(total_amount), 2) AS quarterly_sales
FROM orders
GROUP BY
    YEAR(order_date),
    QUARTER(order_date)
ORDER BY
    order_year,
    order_quarter;


-- 92. Sales by day of week

-- =========================================================
-- 15. FINAL E-COMMERCE ANALYTICS
-- =========================================================

-- 93. Product margin report
SELECT
    product_id,
    product_name,
    cost_price,
    selling_price,
    ROUND(
        selling_price - cost_price,
        2
    ) AS profit_per_unit,
    ROUND(
        (
            selling_price - cost_price
        ) / NULLIF(cost_price, 0) * 100,
        2
    ) AS margin_percentage
FROM products
ORDER BY margin_percentage DESC;


-- 94. Inventory valuation report
SELECT
    p.product_id,
    p.product_name,
    SUM(i.quantity_available) AS total_stock,
    ROUND(
        SUM(i.quantity_available * p.cost_price),
        2
    ) AS inventory_cost_value,
    ROUND(
        SUM(i.quantity_available * p.selling_price),
        2
    ) AS inventory_sales_value
FROM products p
JOIN inventory i
    ON p.product_id = i.product_id
GROUP BY
    p.product_id,
    p.product_name
ORDER BY inventory_sales_value DESC;


-- 95. Product business classification
SELECT
    p.product_id,
    p.product_name,
    p.selling_price,
    p.cost_price,
    ROUND(
        (
            p.selling_price - p.cost_price
        ) / NULLIF(p.cost_price, 0) * 100,
        2
    ) AS margin_percentage,
    CASE
        WHEN p.selling_price >= 50000
             AND p.selling_price - p.cost_price >= 10000
            THEN 'PREMIUM HIGH MARGIN'

        WHEN p.selling_price >= 20000
             AND p.selling_price - p.cost_price >= 5000
            THEN 'HIGH VALUE'

        WHEN p.selling_price - p.cost_price > 0
            THEN 'STANDARD'

        ELSE 'LOW/NO MARGIN'
    END AS business_category
FROM products p
ORDER BY margin_percentage DESC;


-- 96. Customer activity report
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COUNT(o.order_id) AS total_orders,
    MAX(o.order_date) AS latest_order_date,
    COALESCE(
        DATEDIFF(
            CURDATE(),
            MAX(o.order_date)
        ),
        NULL
    ) AS days_since_last_order,
    COALESCE(
        ROUND(SUM(o.total_amount), 2),
        0
    ) AS total_spending
FROM customers c
LEFT JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY total_spending DESC;


-- 97. Low-stock business report
SELECT
    p.product_id,
    p.product_name,
    p.reorder_level,
    SUM(i.quantity_available) AS total_stock,
    CASE
        WHEN SUM(i.quantity_available) = 0
            THEN 'OUT OF STOCK'
        WHEN SUM(i.quantity_available) < p.reorder_level
            THEN 'URGENT REORDER'
        WHEN SUM(i.quantity_available)
             <= p.reorder_level * 2
            THEN 'MONITOR'
        ELSE 'NORMAL'
    END AS stock_status
FROM products p
JOIN inventory i
    ON p.product_id = i.product_id
GROUP BY
    p.product_id,
    p.product_name,
    p.reorder_level
ORDER BY total_stock;


-- 98. Order profitability estimate
SELECT
    o.order_id,
    o.total_amount,
    COALESCE(
        SUM(
            oi.quantity *
            (p.selling_price - p.cost_price)
        ),
        0
    ) AS estimated_gross_profit
FROM orders o
JOIN order_items oi
    ON o.order_id = oi.order_id
JOIN products p
    ON oi.product_id = p.product_id
GROUP BY
    o.order_id,
    o.total_amount
ORDER BY estimated_gross_profit DESC;


-- 99. Product sales and margin analysis
SELECT
    p.product_id,
    p.product_name,
    SUM(oi.quantity) AS units_sold,
    ROUND(SUM(oi.subtotal), 2) AS sales_amount,
    ROUND(
        SUM(
            oi.quantity *
            (p.selling_price - p.cost_price)
        ),
        2
    ) AS estimated_profit
FROM products p
JOIN order_items oi
    ON p.product_id = oi.product_id
GROUP BY
    p.product_id,
    p.product_name
ORDER BY estimated_profit DESC;


-- 100. E-commerce executive summary
SELECT
    (SELECT COUNT(*) FROM customers)
        AS total_customers,

    (SELECT COUNT(*) FROM products)
        AS total_products,

    (SELECT COUNT(*) FROM orders)
        AS total_orders,

    (SELECT ROUND(SUM(total_amount), 2)
     FROM orders)
        AS total_order_value,

    (SELECT ROUND(AVG(total_amount), 2)
     FROM orders)
        AS average_order_value,

    (SELECT COALESCE(SUM(quantity_available), 0)
     FROM inventory)
        AS total_available_stock,

    (SELECT ROUND(SUM(amount), 2)
     FROM payments
     WHERE payment_status = 'SUCCESS')
        AS successful_payment_value,

    (SELECT COALESCE(ROUND(SUM(refund_amount), 2), 0)
     FROM returns)
        AS total_refund_value;
