-- =========================================================
-- 6. AGGREGATES WITH JOINS
-- =========================================================

-- 46. Sales by customer with customer name
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COUNT(o.order_id) AS order_count,
    SUM(o.total_amount) AS total_sales
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY total_sales DESC;


-- 47. Category-wise product count
SELECT
    c.category_name,
    COUNT(p.product_id) AS product_count
FROM categories c
LEFT JOIN products p
    ON c.category_id = p.category_id
GROUP BY c.category_id, c.category_name
ORDER BY product_count DESC;


-- 48. Category-wise average selling price
SELECT
    c.category_name,
    ROUND(AVG(p.selling_price), 2) AS average_selling_price
FROM categories c
JOIN products p
    ON c.category_id = p.category_id
GROUP BY c.category_id, c.category_name
ORDER BY average_selling_price DESC;


-- 49. Category-wise total inventory
SELECT
    c.category_name,
    SUM(i.quantity_available) AS total_available_stock
FROM categories c
JOIN products p
    ON c.category_id = p.category_id
JOIN inventory i
    ON p.product_id = i.product_id
GROUP BY c.category_id, c.category_name
ORDER BY total_available_stock DESC;


-- 50. Supplier-wise product count and average price
SELECT
    s.supplier_name,
    COUNT(p.product_id) AS product_count,
    ROUND(AVG(p.selling_price), 2) AS average_price
FROM suppliers s
JOIN products p
    ON s.supplier_id = p.supplier_id
GROUP BY s.supplier_id, s.supplier_name
ORDER BY product_count DESC;


-- 51. Warehouse-wise inventory
SELECT
    w.warehouse_name,
    SUM(i.quantity_available) AS available_stock,
    SUM(i.quantity_reserved) AS reserved_stock
FROM warehouses w
JOIN inventory i
    ON w.warehouse_id = i.warehouse_id
GROUP BY w.warehouse_id, w.warehouse_name
ORDER BY available_stock DESC;


-- 52. Warehouse-wise number of products
SELECT
    w.warehouse_name,
    COUNT(i.product_id) AS product_count
FROM warehouses w
LEFT JOIN inventory i
    ON w.warehouse_id = i.warehouse_id
GROUP BY w.warehouse_id, w.warehouse_name
ORDER BY product_count DESC;



-- =========================================================
-- 7. ORDER ITEM AGGREGATION
-- =========================================================

-- 53. Total quantity sold for each product
SELECT
    product_id,
    SUM(quantity) AS total_quantity_sold
FROM order_items
GROUP BY product_id
ORDER BY total_quantity_sold DESC;


-- 54. Number of orders containing each product
SELECT
    product_id,
    COUNT(DISTINCT order_id) AS order_count
FROM order_items
GROUP BY product_id
ORDER BY order_count DESC;


-- 55. Total sales generated by each product
SELECT
    product_id,
    SUM(subtotal) AS total_sales
FROM order_items
GROUP BY product_id
ORDER BY total_sales DESC;


-- 56. Product-wise average selling price
SELECT
    product_id,
    ROUND(AVG(unit_price), 2) AS average_selling_price
FROM order_items
GROUP BY product_id
ORDER BY average_selling_price DESC;


-- 57. Product sales with product names
SELECT
    p.product_id,
    p.product_name,
    SUM(oi.quantity) AS total_quantity_sold,
    SUM(oi.subtotal) AS total_sales
FROM products p
JOIN order_items oi
    ON p.product_id = oi.product_id
GROUP BY p.product_id, p.product_name
ORDER BY total_sales DESC;



-- =========================================================
-- 8. HAVING
-- =========================================================

-- 58. Customers who placed more than 2 orders
SELECT
    customer_id,
    COUNT(*) AS order_count
FROM orders
GROUP BY customer_id
HAVING COUNT(*) > 2
ORDER BY order_count DESC;


-- 59. Customers whose total purchase exceeds 50000
SELECT
    customer_id,
    SUM(total_amount) AS total_purchase
FROM orders
GROUP BY customer_id
HAVING SUM(total_amount) > 50000
ORDER BY total_purchase DESC;


-- 60. Products sold more than 10 units
SELECT
    product_id,
    SUM(quantity) AS total_quantity_sold
FROM order_items
GROUP BY product_id
HAVING SUM(quantity) > 10
ORDER BY total_quantity_sold DESC;


-- 61. Categories containing more than 2 products
SELECT
    category_id,
    COUNT(*) AS product_count
FROM products
GROUP BY category_id
HAVING COUNT(*) > 2
ORDER BY product_count DESC;


-- 62. Payment methods with successful payments above 50000
SELECT
    payment_method,
    SUM(amount) AS successful_amount
FROM payments
WHERE payment_status = 'SUCCESS'
GROUP BY payment_method
HAVING SUM(amount) > 50000
ORDER BY successful_amount DESC;



-- =========================================================
-- 9. CONDITIONAL AGGREGATION
-- =========================================================

-- 63. Count orders by status in one result
SELECT
    COUNT(*) AS total_orders,
    SUM(order_status = 'DELIVERED') AS delivered_orders,
    SUM(order_status = 'SHIPPED') AS shipped_orders,
    SUM(order_status = 'PROCESSING') AS processing_orders,
    SUM(order_status = 'CANCELLED') AS cancelled_orders,
    SUM(order_status = 'RETURNED') AS returned_orders
FROM orders;


-- 64. Payment status summary
SELECT
    COUNT(*) AS total_payments,
    SUM(payment_status = 'SUCCESS') AS successful_payments,
    SUM(payment_status = 'FAILED') AS failed_payments,
    SUM(payment_status = 'PENDING') AS pending_payments,
    SUM(payment_status = 'REFUNDED') AS refunded_payments
FROM payments;


-- 65. Inventory stock classification
SELECT
    SUM(quantity_available = 0) AS out_of_stock,
    SUM(quantity_available > 0
        AND quantity_available <= 10) AS low_stock,
    SUM(quantity_available > 10) AS sufficient_stock
FROM inventory;


-- 66. Customer order classification
SELECT
    COUNT(DISTINCT customer_id) AS total_customers,
    COUNT(DISTINCT CASE
        WHEN total_amount >= 10000 THEN customer_id
    END) AS high_value_customers
FROM orders;

    COUNT(*) AS ticket_count

SELECT
    priority,
    ticket_status,
    COUNT(*) AS ticket_count

  AND ticket_status <> 'CLOSED'
GROUP BY priority, ticket_status
ORDER BY ticket_count DESC;



-- =========================================================
-- 14. COMBINED BUSINESS REPORTS
-- =========================================================

-- 83. Customer sales report
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COUNT(o.order_id) AS total_orders,
    COALESCE(SUM(o.total_amount), 0) AS total_spent,
    ROUND(COALESCE(AVG(o.total_amount), 0), 2) AS average_order_value
FROM customers c
LEFT JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY total_spent DESC;


-- 84. Product performance report
SELECT
    p.product_id,
    p.product_name,
    COUNT(DISTINCT oi.order_id) AS orders_count,
    COALESCE(SUM(oi.quantity), 0) AS units_sold,
    COALESCE(SUM(oi.subtotal), 0) AS sales_amount
FROM products p
LEFT JOIN order_items oi
    ON p.product_id = oi.product_id
GROUP BY
    p.product_id,
    p.product_name
ORDER BY sales_amount DESC;


-- 85. Category sales report
SELECT
    c.category_name,
    COUNT(DISTINCT p.product_id) AS product_count,
    COALESCE(SUM(oi.quantity), 0) AS units_sold,
    COALESCE(SUM(oi.subtotal), 0) AS sales_amount
FROM categories c
LEFT JOIN products p
    ON c.category_id = p.category_id
LEFT JOIN order_items oi
    ON p.product_id = oi.product_id
GROUP BY
    c.category_id,
    c.category_name
ORDER BY sales_amount DESC;


-- 86. Warehouse inventory report
SELECT
    w.warehouse_name,
    COUNT(DISTINCT i.product_id) AS product_count,
    SUM(i.quantity_available) AS available_stock,
    SUM(i.quantity_reserved) AS reserved_stock
FROM warehouses w
LEFT JOIN inventory i
    ON w.warehouse_id = i.warehouse_id
GROUP BY
    w.warehouse_id,
    w.warehouse_name
ORDER BY available_stock DESC;


-- 87. Payment collection report
SELECT
    payment_method,
    COUNT(*) AS total_transactions,
    SUM(CASE
        WHEN payment_status = 'SUCCESS' THEN amount
        ELSE 0
    END) AS successful_amount,
    SUM(CASE
        WHEN payment_status = 'REFUNDED' THEN amount
        ELSE 0
    END) AS refunded_amount
FROM payments
GROUP BY payment_method
ORDER BY successful_amount DESC;


-- 88. Order status revenue report
SELECT
    order_status,
    COUNT(*) AS order_count,
    SUM(total_amount) AS total_amount,
    ROUND(AVG(total_amount), 2) AS average_order_value
FROM orders
GROUP BY order_status
ORDER BY total_amount DESC;


-- 89. Top customers by order value
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COUNT(o.order_id) AS total_orders,
    SUM(o.total_amount) AS total_spent
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.first_name,
    c.last_name
ORDER BY total_spent DESC
LIMIT 10;


-- 90. Top-selling products
SELECT
    p.product_id,
    p.product_name,
    SUM(oi.quantity) AS units_sold,
    SUM(oi.subtotal) AS sales_amount
FROM products p
JOIN order_items oi
    ON p.product_id = oi.product_id
GROUP BY
    p.product_id,
    p.product_name
ORDER BY units_sold DESC
LIMIT 10;
