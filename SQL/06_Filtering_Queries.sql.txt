-- ============================================================
-- 1. ACTIVE PRODUCTS ABOVE ₹10,000
-- ============================================================

SELECT
    product_id,
    product_name,
    selling_price,
    status
FROM products
WHERE status = 'ACTIVE'
  AND selling_price > 10000;


-- ============================================================
-- 2. PRODUCTS BETWEEN ₹5,000 AND ₹20,000
-- ============================================================

SELECT
    product_id,
    product_name,
    selling_price
FROM products

ORDER BY selling_price DESC;

-- ============================================================
-- 45. PRODUCTS WHERE PROFIT MARGIN IS ABOVE 20%
-- ============================================================

SELECT
    product_id,
    product_name,
    cost_price,
    selling_price,
    ROUND(
        ((selling_price - cost_price) / selling_price) * 100,
        2
    ) AS profit_margin_percentage
FROM products
WHERE ((selling_price - cost_price) / selling_price) * 100 > 20;

-- ============================================================
-- 52. CASE - PRODUCT PRICE CATEGORY
-- ============================================================

SELECT
    product_id,
    product_name,
    selling_price,
    CASE
        WHEN selling_price < 5000 THEN 'LOW PRICE'
        WHEN selling_price BETWEEN 5000 AND 20000 THEN 'MEDIUM PRICE'
        WHEN selling_price BETWEEN 20001 AND 50000 THEN 'HIGH PRICE'
        ELSE 'PREMIUM PRICE'
    END AS price_category
FROM products;


-- ============================================================
-- 53. CASE - ORDER VALUE CATEGORY
-- ============================================================

SELECT
    order_id,
    total_amount,
    CASE
        WHEN total_amount < 5000 THEN 'LOW VALUE'
        WHEN total_amount BETWEEN 5000 AND 20000 THEN 'MEDIUM VALUE'
        WHEN total_amount BETWEEN 20001 AND 50000 THEN 'HIGH VALUE'
        ELSE 'PREMIUM VALUE'
    END AS order_value_category
FROM orders;


-- ============================================================
-- 54. CASE - PAYMENT STATUS
-- ============================================================

SELECT
    payment_id,
    order_id,
    amount,
    payment_status,
    CASE
        WHEN payment_status = 'SUCCESS' THEN 'COMPLETED'
        WHEN payment_status = 'PENDING' THEN 'WAITING'
        WHEN payment_status = 'FAILED' THEN 'ACTION REQUIRED'
        WHEN payment_status = 'REFUNDED' THEN 'REFUND COMPLETED'
        ELSE 'UNKNOWN'
    END AS payment_category
FROM payments;


-- ============================================================
-- 55. CASE - STOCK STATUS
-- ============================================================

SELECT
    i.product_id,
    p.product_name,
    i.quantity_available,
    p.reorder_level,
    CASE
        WHEN i.quantity_available = 0 THEN 'OUT OF STOCK'
        WHEN i.quantity_available <= p.reorder_level THEN 'LOW STOCK'
        ELSE 'AVAILABLE'
    END AS stock_status
FROM inventory i
JOIN products p
    ON i.product_id = p.product_id;


-- ============================================================
-- 56. COALESCE - DISPLAY ZERO FOR NULL DISCOUNTS
-- ============================================================

SELECT
    order_id,
    subtotal,
    COALESCE(discount_amount, 0) AS discount_amount,
    total_amount
FROM orders;


-- ============================================================
-- 57. COALESCE - DISPLAY "NOT DELIVERED"
-- ============================================================

SELECT
    shipment_id,
    order_id,
    shipped_at,
    COALESCE(
        DATE_FORMAT(delivered_at, '%Y-%m-%d %H:%i:%s'),
        'NOT DELIVERED'
    ) AS delivery_date
FROM shipments;


-- ============================================================
-- 58. STRING FUNCTION - UPPERCASE PRODUCT NAMES
-- ============================================================

SELECT

-- ============================================================
-- 59. STRING FUNCTION - LOWERCASE EMAILS
-- ============================================================

SELECT
    customer_id,

-- ============================================================
-- 60. STRING LENGTH
-- ============================================================

SELECT
    product_id,
    product_name,
    LENGTH(product_name) AS name_length
FROM products

-- ============================================================
-- 61. CONCAT CUSTOMER FULL NAME
-- ============================================================

SELECT
    customer_id,

    email
FROM customers;


-- ============================================================
-- 62. DATE DIFFERENCE - DELIVERY TIME
-- ============================================================

SELECT
    shipment_id,
    order_id,
    shipped_at,
    delivered_at,
    DATEDIFF(delivered_at, shipped_at) AS delivery_days
FROM shipments
WHERE delivered_at IS NOT NULL;


-- ============================================================
-- 63. DELIVERY DELAY ABOVE 2 DAYS
-- ============================================================

SELECT
    shipment_id,
    order_id,
    courier_name,
    DATEDIFF(delivered_at, shipped_at) AS delivery_days
FROM shipments
WHERE delivered_at IS NOT NULL
  AND DATEDIFF(delivered_at, shipped_at) > 2;

-- ============================================================
-- 66. PRODUCTS SORTED BY PROFIT
-- ============================================================

SELECT
    product_id,
    product_name,
    cost_price,
    selling_price,
    selling_price - cost_price AS profit_per_unit
FROM products
ORDER BY profit_per_unit DESC;


-- ============================================================
-- 67. TOP 5 PRODUCTS BY PROFIT
-- ============================================================

SELECT
    product_id,
    product_name,
    selling_price - cost_price AS profit_per_unit
FROM products
ORDER BY profit_per_unit DESC
LIMIT 5;


-- ============================================================
-- 68. ORDERS REQUIRING ATTENTION
-- FAILED PAYMENT OR CANCELLED ORDER
-- ============================================================

SELECT
    o.order_id,
    o.order_status,
    o.total_amount,
    p.payment_status
FROM orders o
LEFT JOIN payments p
    ON o.order_id = p.order_id
WHERE o.order_status = 'CANCELLED'
   OR p.payment_status = 'FAILED';


-- ============================================================
-- 69. ORDERS WITH PENDING PAYMENT
-- ============================================================

SELECT
    o.order_id,
    o.customer_id,
    o.order_status,
    o.total_amount,
    p.payment_method,
    p.payment_status
FROM orders o
JOIN payments p
    ON o.order_id = p.order_id
WHERE p.payment_status = 'PENDING';


-- ============================================================
-- 70. BUSINESS FILTER:
-- DELIVERED ORDERS ABOVE ₹20,000
-- ============================================================

SELECT
    order_id,
    customer_id,
    order_date,
    total_amount
FROM orders
WHERE order_status = 'DELIVERED'
  AND total_amount > 20000
ORDER BY total_amount DESC;
